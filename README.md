document.getElementById("rsvpForm").addEventListener("submit", function(event) {
  event.preventDefault();
  const name = document.getElementById("name").value.trim();
  const attendance = document.querySelector('input[name="attendance"]:checked');
  if (!name || !attendance) return;
  const message = attendance.value === "Yes"
    ? `Thank you, ${name}! Your RSVP has been prepared. Please note: this demo does not send your response to the organizers.`
    : `Thank you for letting us know, ${name}. This demo does not send your response to the organizers.`;
  const box = document.getElementById("confirmation");
  box.textContent = message;
  box.style.display = "block";
});
